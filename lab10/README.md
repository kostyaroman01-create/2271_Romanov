* & - взятия адреса
* первая программа
* во второй строчке * указатель, а & указывает на адрес а
* в третьей строчке q присваивается указатель p
* четвертая строчка * берет число с указателя

* вторая программа
* в третьей строчке *q обращается к указателю а и присваивает его
```

#include <vector>

   {
   int a = 10, b = 20;
   int *p = &a , *q = &b;
   p = q;
   ;
   std::cout << "a " << a <<' ' << "b " << b  << "p " << *p <<' '<< "q " << *q << std::endl;
   }
   {
   int a = 10, b = 20;
   int *p = &a , *q = &b;
   *p = *q;
   std::cout << "a " << a <<' ' << "b " << b  << "p " << *p <<' '<< "q " << *q << std::endl;
   }
   return 0;

}
```
